<script type="text/babel">
const { useState } = React;

function App() {
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [userRole, setUserRole] = useState('');

  const [tripData, setTripData] = useState({
    date: '', village: '', wheatBags: 0, riceBags: 0, sugarBags: 0, otherBags: 0,
    remainingBags: 0, freightEarned: 0, expenseAmount: 0, expenseReason: ''
  });

  const [ledger, setLedger] = useState([
    { id: 1, date: '2026-08-20', village: 'Palam Market', wheatBags: 150, riceBags: 100, sugarBags: 50, otherBags: 0, totalBags: 300, remainingBags: 10, freightEarned: 15000, expenseAmount: 2000, expenseReason: 'Diesel & Toll' }
  ]);

  const handleLogin = (e) => {
    e.preventDefault();
    if (username === 'admin' && password === 'Admin@123') {
      setUserRole('District Admin');
      setIsLoggedIn(true);
    } else if (username === 'operator' && password === 'Operator@123') {
      setUserRole('Palam Operator');
      setIsLoggedIn(true);
    } else {
      alert('Galat Username ya Password!');
    }
  };

  const handleAddTrip = (e) => {
    e.preventDefault();
    const totalBags = Number(tripData.wheatBags) + Number(tripData.riceBags) + Number(tripData.sugarBags) + Number(tripData.otherBags);
    const newEntry = {
      id: Date.now(),
      ...tripData,
      totalBags,
      freightEarned: Number(tripData.freightEarned),
      expenseAmount: Number(tripData.expenseAmount),
      remainingBags: Number(tripData.remainingBags)
    };
    setLedger([newEntry, ...ledger]);
    alert('Trip Entry Save Ho Gayi!');
  };

  const totalFreight = ledger.reduce((acc, item) => acc + item.freightEarned, 0);
  const totalExpenses = ledger.reduce((acc, item) => acc + item.expenseAmount, 0);
  const netProfit = totalFreight - totalExpenses;

  if (!isLoggedIn) {
    return (
      <div class="login-container">
        <div class="card" style={{width: '320px', textAlign: 'center'}}>
          <h2>FC CONTRACTOR & TRANSPORT</h2>
          <p style={{color: '#666'}}>Palam, Parbhani</p>
          <form onSubmit={handleLogin}>
            <input type="text" placeholder="Username" value={username} onChange={e=>setUsername(e.target.value)} required />
            <input type="password" placeholder="Password" value={password} onChange={e=>setPassword(e.target.value)} required />
            <button type="submit">Login</button>
          </form>
        </div>
      </div>
    );
  }

  return (
    <div>
      <div class="header">
        <div>
          <h2 style={{margin:0}}>FC CONTRACTOR & TRANSPORT</h2>
          <small>Role: {userRole}</small>
        </div>
        <button class="btn-logout" onClick={()=>setIsLoggedIn(false)}>Logout</button>
      </div>

      <div style={{padding: '20px', maxWidth: '1200px', margin: '0 auto'}}>
        <div class="card">
          <h3>New Trip Entry</h3>
          <form onSubmit={handleAddTrip} class="grid-form">
            <input type="date" required value={tripData.date} onChange={e=>setTripData({...tripData, date: e.target.value})} />
            <input type="text" placeholder="Village Name" required value={tripData.village} onChange={e=>setTripData({...tripData, village: e.target.value})} />
            <input type="number" placeholder="Wheat Bags" onChange={e=>setTripData({...tripData, wheatBags: e.target.value})} />
            <input type="number" placeholder="Rice Bags" onChange={e=>setTripData({...tripData, riceBags: e.target.value})} />
            <input type="number" placeholder="Sugar Bags" onChange={e=>setTripData({...tripData, sugarBags: e.target.value})} />
            <input type="number" placeholder="Other Bags" onChange={e=>setTripData({...tripData, otherBags: e.target.value})} />
            <input type="number" placeholder="Remaining Bags" onChange={e=>setTripData({...tripData, remainingBags: e.target.value})} />
            <input type="number" placeholder="Freight Earned (â¹)" onChange={e=>setTripData({...tripData, freightEarned: e.target.value})} />
            <input type="number" placeholder="Expense Amount (â¹)" onChange={e=>setTripData({...tripData, expenseAmount: e.target.value})} />
            <input type="text" placeholder="Expense Reason" onChange={e=>setTripData({...tripData, expenseReason: e.target.value})} />
            <button type="submit" style={{gridColumn: '1 / -1'}}>Save Entry</button>
          </form>
        </div>

        <div class="card" style={{overflowX: 'auto'}}>
          <h3>Monthly Vehicle Ledger</h3>
          <table>
            <thead>
              <tr>
                <th>S.No</th><th>Date</th><th>Village</th><th>Wheat</th><th>Rice</th><th>Sugar</th><th>Other</th><th>Total Bags</th><th>Remaining</th><th>Freight (â¹)</th><th>Expense (â¹)</th><th>Reason</th>
              </tr>
            </thead>
            <tbody>
              {ledger.map((item, idx)=>(
                <tr key={item.id}>
                  <td>{idx+1}</td><td>{item.date}</td><td>{item.village}</td><td>{item.wheatBags}</td><td>{item.riceBags}</td><td>{item.sugarBags}</td><td>{item.otherBags}</td><td><b>{item.totalBags}</b></td><td style={{color:'red'}}>{item.remainingBags}</td><td style={{color:'green'}}>â¹{item.freightEarned}</td><td style={{color:'orange'}}>â¹{item.expenseAmount}</td><td>{item.expenseReason}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>

        <div class="summary-grid">
          <div class="summary-card" style={{borderColor: 'green'}}>
            <h4>Total Freight</h4>
            <h2>â¹{totalFreight}</h2>
          </div>
          <div class="summary-card" style={{borderColor: 'orange'}}>
            <h4>Total Expenses</h4>
            <h2>â¹{totalExpenses}</h2>
          </div>
          <div class="summary-card" style={{borderColor: 'blue'}}>
            <h4>Net Profit</h4>
            <h2>â¹{netProfit}</h2>
          </div>
        </div>
      </div>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App />);
</script>
